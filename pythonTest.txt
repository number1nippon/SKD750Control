orientation_lock = threading.Lock()


# Nikon orientation mapping:
#
# Internal value -> physical camera orientation
#
# 0 =   0 degrees
# 1 = 270 degrees
# 2 =  90 degrees
# 3 = 180 degrees

camera_orientation = 0


def orientation_degrees(value):

    return {
        0: 0,
        1: 270,
        2: 90,
        3: 180
    }.get(value, 0)


def degrees_to_orientation(value):

    value = value % 360

    return {
        0: 0,
        270: 1,
        90: 2,
        180: 3
    }.get(value)


def set_camera_orientation(value):

    global camera_orientation

    if value not in (0, 1, 2, 3):

        return

    with orientation_lock:

        if camera_orientation != value:

            print(
                "Camera orientation changed:",
                orientation_degrees(
                    camera_orientation
                ),
                "->",
                orientation_degrees(
                    value
                )
            )

            camera_orientation = value


def get_camera_orientation():

    with orientation_lock:

        return camera_orientation


# ==================================================
# CAMERA EVENT LISTENER
# ==================================================

def parse_orientation_event(event_data):

    if event_data is None:

        return None

    text = str(event_data)

    if "d10e" not in text.lower():

        return None

    match = re.search(
        r'orientation.*?to\s+"?(-?\d+)',
        text,
        re.IGNORECASE
    )

    if not match:

        return None

    degrees = int(
        match.group(1)
    )

    return degrees_to_orientation(
        degrees
    )


def camera_event_loop():

    print(
        "Camera event listener started."
    )

    while True:

        try:

            with camera_lock:

                event_type, event_data = (
                    camera.wait_for_event(
                        1
                    )
                )

            if event_data is None:

                continue

            new_orientation = (
                parse_orientation_event(
                    event_data
                )
            )

            if new_orientation is not None:

                set_camera_orientation(
                    new_orientation
                )

                print(
                    "Orientation event:",
                    event_data
                )

        except Exception as error:

            print(
                "Camera event error:",
                error
            )

            time.sleep(
                0.01
            )

def initialise_camera():

    global camera

    print(
        "Initialising camera..."
    )

    camera = gp.Camera()

    camera.init()

    print(
        "Camera connected."
    )

    config = camera.get_config()

    viewfinder = (
        config.get_child_by_name(
            "viewfinder"
        )
    )

        )

    viewfinder.set_value(
        1
    )

    camera.set_config(
        config
    )

    print(
        "Live View started."
    )

def get_camera_widget(
    config,
    name
):

    widget = (
        config.get_child_by_name(
            name
        )
    )

    if widget is None:

        raise RuntimeError(
            "Camera control not available: "
            +
            name
        )

    return widget


def get_setting(
    name
):

    config = camera.get_config()

    widget = get_camera_widget(
        config,
        name
    )

    return widget


def get_widget_choices(widget):

    choices = []

    try:

        count = widget.count_choices()

        for index in range(count):

            choices.append(
                widget.get_choice(
                    index
                )
            )

        return choices

    except Exception:

        pass


    #
    # Some versions/bindings expose
    # the choices differently.
    #

    try:

        choices = widget.get_choices()

        if choices is not None:

            return list(
                choices
            )

    except Exception:

        pass


    return []


def get_setting_state(
    name
):

    config = camera.get_config()

    widget = get_camera_widget(
        config,
        name
    )

    value = widget.get_value()

    choices = get_widget_choices(
        widget
    )

    return {
        "value": str(value),
        "choices": choices
    }


def change_choice_setting(
    name,
    direction
):

    config = camera.get_config()

    widget = get_camera_widget(
        config,
        name
    )

    choices = get_widget_choices(
        widget
    )

    if not choices:

        raise RuntimeError(
            "No choices available for "
            +
            name
        )


    current = str(
        widget.get_value()
    )


    try:

        current_index = (
            choices.index(
                current
            )
        )

    except ValueError:

        #
        # Try matching by string representation.
        #

        current_index = -1

        for index, choice in enumerate(
            choices
        ):

            if str(choice) == current:

                current_index = index

                break


    if current_index < 0:

        raise RuntimeError(
            "Current value '"
            +
            current
            +
            "' was not found in "
            +
            name
            +
            " choices"
        )


    new_index = (
        current_index
        +
        direction
    )


    #
    # Do not wrap around.
    #

    if new_index < 0:

        new_index = 0

    if new_index >= len(choices):

        new_index = (
            len(choices) - 1
        )


    new_value = choices[
        new_index
    ]


    print(
        "Changing",
        name,
        ":",
        current,
        "->",
        new_value
    )


    widget.set_value(
        new_value
    )

    camera.set_config(
        config
    )


    #
    # Read it back from the camera.
    #

    config = camera.get_config()

    widget = get_camera_widget(
        config,
        name
    )

    actual = widget.get_value()


    return {
        "value": str(actual),
        "choices": get_widget_choices(
            widget
        )
    }

def get_exposure_preview():

    config = camera.get_config()

    widget = (
        config.get_child_by_name(

        )
    )

    if widget is None:

        raise RuntimeError(
            "Nikon Exposure Preview control d1a5 "
            "is not available in this libgphoto2 configuration"
        )

    value = widget.get_value()

    print(
        "Exposure Preview current value:",
        value
    )

    return str(value) == "1"


def set_exposure_preview(enabled):

    global exposure_preview

    config = camera.get_config()

    widget = (
        config.get_child_by_name(
            "d1a5"
        )
    )

    if widget is None:

        raise RuntimeError(
            "Nikon Exposure Preview control d1a5 "
            "is not available in this libgphoto2 configuration"
        )

    value = (
        "1"
        if enabled
        else "0"
    )

    print(
        "Setting Exposure Preview to:",
        value
    )

    widget.set_value(
        value
    )

    camera.set_config(
        config
    )

    actual = (
        get_exposure_preview()
    )

    exposure_preview = actual

    return actual

EXPOSURE_SETTINGS = {

    "shutter": {
        "widget": "shutterspeed",
        "label": "Shutter"
    },

    "aperture": {
        "widget": "f-number",
        "label": "Aperture"
    },

        "label": "ISO"
    },

    "exposure_compensation": {
        "widget": "exposurecompensation",
        "label": "Exposure Compensation"
    }

def read_exposure_settings():

    result = {}

    config = camera.get_config()

    for key, info in (
        EXPOSURE_SETTINGS.items()
    ):

        try:

            widget = get_camera_widget(
                config,
                info["widget"]
            )

            result[key] = {
                "value": str(
                    widget.get_value()
                ),
                "choices":
                    get_widget_choices(
                        widget
                    )
            }

        except Exception as error:

            result[key] = {
                "value": None,
                "choices": [],
                "error": str(error)
            }


    return result

def clamp(
    value,
    minimum,
    maximum
):

    return max(
        minimum,
        min(
            maximum,
            value
        )
    )


def normalised_to_sensor(
    normal_x,
    normal_y,
    orientation
):

    #
    # normal_x / normal_y represent
    # coordinates in the ORIGINAL,
    # UNROTATED 640 x 480 live-view frame.
    #
    # The browser reverses display rotation.
    #
    # Therefore the sensor mapping is direct.
    #

    sensor_x = int(
        normal_x * SENSOR_WIDTH
    )

    sensor_y = int(
        normal_y * SENSOR_HEIGHT
    )

    sensor_x = clamp(
        sensor_x,
        0,
        SENSOR_WIDTH - 1
    )

    sensor_y = clamp(
        sensor_y,
        0,
        SENSOR_HEIGHT - 1
    )

    return (
        sensor_x,
        sensor_y
    )

@app.route(
    "/focus",
    methods=["POST"]
)
def focus():

    try:

        data = request.get_json(
            force=True
        )

        normal_x = float(
            data.get(
                "x"
            )
        )

        normal_y = float(
            data.get(
                "y"
            )
        )


        normal_x = clamp(
            normal_x,
            0.0,
            1.0
        )

        normal_y = clamp(
            normal_y,
            0.0,
            1.0
        )


        orientation = (
            get_camera_orientation()
        )


        sensor_x, sensor_y = (
            normalised_to_sensor(
                normal_x,
                normal_y,
                orientation_degrees(
                    orientation
                )
            )
        )


        print(
            "Focus request:"
        )

        print(
            "  Browser X:",
            normal_x
        )

        print(
            "  Browser Y:",
            normal_y
        )

        print(
            "  Orientation:",
            orientation
        )

        print(
            "  Orientation degrees:",
            orientation_degrees(
                orientation
            )
        )


        with camera_lock:

            set_af_area(
                sensor_x,
                sensor_y
            )

            time.sleep(
                0.1
            )

            drive_autofocus()


        return jsonify(
            {
                "success": True,
                "normal_x": normal_x,
                "normal_y": normal_y,
                "sensor_x": sensor_x,
                "sensor_y": sensor_y,
                "orientation": orientation,
                "degrees": orientation_degrees(
                    orientation
                )
            }
        )


    except Exception as error:

        print(
            "Focus error:",
            error
        )

        return jsonify(
            {
                "success": False,
                "error": str(error)
            }
        ), 500

def camera_stream():

    while True:

        try:

            with camera_lock:

                preview = (
                    camera.capture_preview()
                )

                data = (
                    preview.get_data_and_size()
                )

                jpeg = bytes(
                    data
                )


            yield (
                b"--"
                +
                BOUNDARY
                +
                b"\r\n"
                +
                b"Content-Type: image/jpeg\r\n"
                +
                b"Content-Length: "
                +
                str(
                    len(jpeg)
                ).encode()
                +
                b"\r\n\r\n"
                +
                jpeg
                +
                b"\r\n"
            )


        except GeneratorExit:

            break


        except Exception as error:

            print(
                "Live View error:",
                error
            )

            time.sleep(
                0.01
            )


@app.route(
    "/camera"
)
def camera_feed():

    return Response(
        camera_stream(),
        content_type=
            "multipart/x-mixed-replace; boundary=frame"
    )

   ORIENTATION

function updateOrientation(
    orientation
) {

    const status =
        document.getElementById(
            "orientationStatus"
        );


    let degrees = 0;


    if (
        orientation === 0
    ) {

        degrees = 0;

    } else if (
        orientation === 1
    ) {

        degrees = 270;

    } else if (
        orientation === 2
    ) {

        degrees = 90;

    } else if (
        orientation === 3
    ) {

        degrees = 180;
    }


    status.innerText =
        "Orientation: "
        +
        degrees
        +
        "°";


    if (
        currentOrientation ===
        orientation
    ) {

        return;
    }


    currentOrientation =
        orientation;


    const camera =
        document.getElementById(
            "camera"
        );


    if (
        orientation === 1 ||
        orientation === 2
    ) {

        camera.style.transform =

            orientation === 2

                ? "rotate(90deg)"

                : "rotate(-90deg)";

    } else if (
        orientation === 3
    ) {

        camera.style.transform =
            "rotate(180deg)";

    } else {

        camera.style.transform =
            "rotate(0deg)";
    }


    requestAnimationFrame(
        positionFocusBox
    );
}

function positionFocusBox() {

    if (
        focusNormalX === null ||
        focusNormalY === null
    ) {

        return;

    const camera =
        document.getElementById(
            "camera"
        );


    const focusBox =
        document.getElementById(
            "focusBox"
        );


    const rect =
        camera.getBoundingClientRect();


    const x =
        rect.left
        +
        (
            focusNormalX
            *
            rect.width
        );


    const y =
        rect.top
        +
        (
            focusNormalY
            *
            rect.height
        );


    focusBox.style.left =
        x + "px";


    focusBox.style.top =
        y + "px";


    focusBox.style.display =
        "block";
}

function clickToFocus(
    event
) {

    const camera =
        document.getElementById(
            "camera"
        );


    const rect =
        camera.getBoundingClientRect();


    let displayX =
        (
            event.clientX -
            rect.left
        ) / rect.width;


    let displayY =
        (
            event.clientY -
            rect.top
        ) / rect.height;


    if (
        displayX < 0 ||
        displayX > 1 ||
        displayY < 0 ||
        displayY > 1
    ) {

        return;
    }


    let normalX;
    let normalY;


    if (
        currentOrientation === 0
    ) {

        /*
         * 0° — LANDSCAPE
         */

        normalX =
            displayX;

        normalY =
            displayY;


    } else if (
        currentOrientation === 1
    ) {

        /*
         * 270° — UPRIGHT PORTRAIT
         */

        normalX =
            1.0 - displayY;

        normalY =
            displayX;


    } else if (
        currentOrientation === 2
    ) {

        /*
         * 90° — UPSIDE-DOWN PORTRAIT
         */

        normalX =
            displayY;

        normalY =
            1.0 - displayX;


    } else if (
        currentOrientation === 3
    ) {

        /*
         * 180° — UPSIDE-DOWN LANDSCAPE
         */

        normalX =
            1.0 - displayX;

        normalY =
            1.0 - displayY;


    } else {

        normalX =
            displayX;

        normalY =
            displayY;
    }


    focusNormalX =
        displayX;

    focusNormalY =
        displayY;


    positionFocusBox();


    const status =
        document.getElementById(
            "focusStatus"
        );


    status.innerText =
        "Focusing...";


    fetch(
        "/focus",
        {

            method:
                "POST",

            headers:
                {
                    "Content-Type":
                        "application/json"
                },

            body:
                JSON.stringify(
                    {
                        x:
                            normalX,

                        y:
                            normalY
                    }
                )
        }
    )

    .then(
        response =>
            response.json()
    )

    .then(
        data => {

            if (
                data.success
            ) {

                status.innerText =

                    "Focus: "

                    +

                    data.sensor_x

                    +

                    ", "

                    +

                    data.sensor_y

                    +

                    " @ "

                    +

                    data.degrees

                    +

                    "°";

            } else {

                status.innerText =
                    "Focus error";

                console.error(
                    data.error
                );
            }

        }
    )

    .catch(
        error => {

            status.innerText =
                "Focus error";

            console.error(
                error
            );

        }
    );
}
